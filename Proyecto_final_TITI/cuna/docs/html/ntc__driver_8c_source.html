122float leer_temperatura_celsius(adc_oneshot_unit_handle_t manejador_adc) {
123 int valor_adc_raw;
124
125 // PASO 1: Leer valor digital del ADC (0 a 4095)
126 adc_oneshot_read(manejador_adc, CANAL_ADC_NTC, &valor_adc_raw);
127
128 // PASO 2: Convertir valor digital a voltaje analógico
129 // Fórmula: V = (ADC / ADC_max) * V_ref
130 float voltaje_mv = (valor_adc_raw * VOLTAJE_REFERENCIA_MV) / 4095.0f;
131
132 // Validación: Detectar circuito abierto (NTC desconectado)
133 // Si V ≥ V_ref, no hay caída de voltaje en R_div → circuito abierto
134 if (voltaje_mv >= VOLTAJE_REFERENCIA_MV) {
135 return -999.0f; // Código de error: circuito abierto
136 }
137
138 // PASO 3: Calcular resistencia del NTC usando divisor de voltaje
139 // Del divisor de voltaje: V_out = V_in * (R_ntc / (R_div + R_ntc))
140 // Despejando R_ntc: R_ntc = R_div * V_out / (V_in - V_out)
141 float resistencia_ntc = RESISTENCIA_DIVISOR * voltaje_mv / (VOLTAJE_REFERENCIA_MV - voltaje_mv);
142
143 // Validación: Detectar cortocircuito o error de cálculo
144 if (resistencia_ntc <= 0) {
145 return -999.0f; // Código de error: resistencia inválida
146 }
147
148 // PASO 4: Aplicar ecuación de Steinhart-Hart simplificada con Beta
149 // Fórmula: 1/T = 1/T0 + (1/B) * ln(R/R0)
150 // Donde: T y T0 en Kelvin, B en Kelvin, R y R0 en Ohms
151 float steinhart = log(resistencia_ntc / RESISTENCIA_NOMINAL_NTC) / COEFICIENTE_BETA_NTC;
152 steinhart += 1.0 / (TEMP_NOMINAL_NTC + 273.15); // Suma 1/T0 (T0 en Kelvin)
153
154 // PASO 5: Convertir de Kelvin a Celsius
155 // T_celsius = T_kelvin - 273.15
156 // Nota: steinhart contiene 1/T, por eso se invierte primero
157 return (1.0 / steinhart) - 273.15;
158}
