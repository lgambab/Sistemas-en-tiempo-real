118static void sensor_update_task(void* arg) {
119 ESP_LOGI(TAG, "Sensor task started");
120
121 while (1) {
122 sensor_data_t local_data = {0};
123
124 // Leer temperatura
125 if (adc_handle) {
126 local_data.temperature_celsius = leer_temperatura_celsius(adc_handle);
127 local_data.temperature_valid = (local_data.temperature_celsius > -100);
128 } else {
129 local_data.temperature_valid = false;
130 }
131
132 // Leer hora actual
133 time_t now;
134 time(&now);
135 localtime_r(&now, &local_data.time_info);
136 local_data.time_valid = (local_data.time_info.tm_year > (2000 - 1900));
137
138 // Actualizar datos compartidos de forma segura
139 if (xSemaphoreTake(sensor_mutex, pdMS_TO_TICKS(100)) == pdTRUE) {
140 shared_sensor_data = local_data;
141 xSemaphoreGive(sensor_mutex);
142 }
143
144 // Actualizar cada 2 segundos
145 vTaskDelay(pdMS_TO_TICKS(2000));
146 }
147}

175esp_err_t sensor_task_init(adc_oneshot_unit_handle_t handle) {
176 adc_handle = handle;
177
178 // Crear mutex
179 sensor_mutex = xSemaphoreCreateMutex();
180 if (!sensor_mutex) {
181 ESP_LOGE(TAG, "Failed to create mutex");
182 return ESP_ERR_NO_MEM;
183 }
184
185 // Crear tarea de sensores
186 BaseType_t ret = xTaskCreate(
187 sensor_update_task,
188 "sensor_task",
189 3072,
190 NULL,
191 configMAX_PRIORITIES - 3, // Prioridad media
192 NULL
193 );
194
195 if (ret != pdPASS) {
196 ESP_LOGE(TAG, "Failed to create sensor task");
197 vSemaphoreDelete(sensor_mutex);
198 return ESP_FAIL;
199 }
200
201 ESP_LOGI(TAG, "Sensor task initialized");
202 return ESP_OK;
203}

244esp_err_t sensor_task_get_data(sensor_data_t* data) {
245 if (!data) return ESP_ERR_INVALID_ARG;
246
247 // Verificar que el mutex existe antes de usarlo
248 if (!sensor_mutex) {
249 ESP_LOGW(TAG, "Sensor task not initialized");
250 memset(data, 0, sizeof(*data));
251 return ESP_ERR_INVALID_STATE;
252 }
253
254 if (xSemaphoreTake(sensor_mutex, pdMS_TO_TICKS(100)) == pdTRUE) {
255 *data = shared_sensor_data;
256 xSemaphoreGive(sensor_mutex);
257 return ESP_OK;
258 }
259
260 ESP_LOGW(TAG, "Failed to acquire mutex");
261 return ESP_ERR_TIMEOUT;
262}
