103static uint8_t compute_auto_speed(float temp_c)
104{
105 /*
106 Reglas:
107 - < 25°C => 0%
108 - 25–30°C => proporcional
109 - > 30°C => 100%
110 */
111
112 if (temp_c < 25.0f) {
113 return 0;
114 }
115
116 if (temp_c >= 30.0f) {
117 return 100;
118 }
119
120 // Mapeo lineal 25 → 0% 30 → 100%
121 float scale = (temp_c - 25.0f) / 5.0f; // 0 a 1
122 return (uint8_t)(scale * 100.0f);
123}

184void fan_control_set_mode(fan_mode_t mode)
185{
186 s_mode = mode;
187
188 switch (s_mode) {
189
190 case FAN_MODE_MANUAL:
191 // usa directamente la velocidad del slider
192 s_current_speed = s_manual_speed;
193 fan_set_speed_percent(s_current_speed);
194 break;
195
196 case FAN_MODE_AUTO: {
197 // Al cambiar a AUTO, aplicamos una primera vez lógica NTC+PIR
198 if (s_ntc_adc_handle == NULL) {
199 s_current_speed = 0;
200 fan_off();
201 break;
202 }
203
204 float temp_c = leer_temperatura_celsius(s_ntc_adc_handle);
205 if (temp_c < -100.0f) { // error NTC
206 s_current_speed = 0;
207 fan_off();
208 break;
209 }
210
211 bool motion = pir_is_motion_active();
212 if (!motion) {
213 s_current_speed = 0;
214 fan_off();
215 break;
216 }
217
218 uint8_t spd = compute_auto_speed(temp_c);
219 s_current_speed = spd;
220 fan_set_speed_percent(s_current_speed);
221 break;
222 }
223
224 case FAN_MODE_REGISTERS:
225 if (s_reg_active) {
226 s_current_speed = s_manual_speed;
227 fan_set_speed_percent(s_current_speed);
228 } else {
229 s_current_speed = 0;
230 fan_off();
231 }
232 break;
233 }
234}

318void fan_control_on_register_tick(bool any_match_now)
319{
320 s_reg_active = any_match_now;
321
322 // Solo tiene efecto en modo REGISTROS
323 if (s_mode != FAN_MODE_REGISTERS) {
324 return;
325 }
326
327 if (s_reg_active) {
328 // Aquí decides a qué velocidad se enciende cuando un registro coincide.
329 // Yo uso la velocidad del slider (s_manual_speed).
330 s_current_speed = s_manual_speed;
331 fan_set_speed_percent(s_current_speed);
332 } else {
333 // Ningún registro activo → ventilador apagado
334 s_current_speed = 0;
335 fan_off();
336 }
337}
