96/**
97 * @brief Ejecutar actualización de firmware via OTA
98 * @details Envía el archivo .bin seleccionado al ESP32 mediante POST a /OTAupdate.
99 * Muestra progreso de carga y maneja la respuesta del servidor.

101 * @note El ESP32 debe reiniciarse automáticamente tras aplicar el nuevo firmware

104function updateFirmware()
105{
106 // Form Data
107 var formData = new FormData();
108 var fileSelect = document.getElementById("selected_file");
109
110 if (fileSelect.files && fileSelect.files.length == 1)
111 {
112 var file = fileSelect.files[0];

114 document.getElementById("ota_update_status").innerHTML = "Uploading " + file.name + ", Firmware Update in Progress...";
115
116 // Http Request
117 var request = new XMLHttpRequest();
118
119 request.upload.addEventListener("progress", updateProgress);
120 request.open('POST', "/OTAupdate");
121 request.responseType = "blob";
122 request.send(formData);
123 }
124 else
125 {
126 window.alert('Select A File First')
127 }
128}

147/**
148 * @brief Consultar estado de actualización OTA
149 * @details Realiza POST a /OTAstatus y procesa la respuesta JSON:
150 * - ota_update_status = 1: Actualización exitosa, inicia temporizador de reinicio
151 * - ota_update_status = -1: Error en la actualización
152 * - ota_update_status = 0: Información sobre firmware actual
153 *
154 * @note También actualiza #latest_firmware con fecha/hora de compilación
155 */
156function getUpdateStatus()
157{
158 var xhr = new XMLHttpRequest();
159 var requestURL = "/OTAstatus";
160 xhr.open('POST', requestURL, false);
161 xhr.send('ota_update_status');
162
163 if (xhr.readyState == 4 && xhr.status == 200)
164 {
165 var response = JSON.parse(xhr.responseText);
166
167 document.getElementById("latest_firmware").innerHTML = response.compile_date + " - " + response.compile_time
168
169 // If flashing was complete it will return a 1, else -1
170 // A return of 0 is just for information on the Latest Firmware request
171 if (response.ota_update_status == 1)
172 {
173 // Set the countdown timer time
174 seconds = 10;
175 // Start the countdown timer
176 otaRebootTimer();
177 }
178 else if (response.ota_update_status == -1)
179 {
180 document.getElementById("ota_update_status").innerHTML = "!!! Upload Error !!!";
181 }
182 }
183}

197 window.location.reload();
198 }
199 else
200 {

236/**
237 * @brief Leer valores de sensores (temperatura NTC y PIR)
238 * @details Realiza GET a /dhtSensor.json y actualiza:
239 * - #temperature_reading: Temperatura en °C del sensor NTC
240 * - #pir_status: Estado del sensor PIR (movimiento detectado / sin movimiento)
241 *
242 * Maneja valores nulos y errores de lectura mostrando "--" o "Sin datos"
243 */
244function getDHTSensorValues()
245{
246 $.getJSON('/dhtSensor.json', function(data) {
247
248 // ----- Temperatura NTC -----
249 if (data["temp"] === null || data["temp"] === undefined) {
250 $("#temperature_reading").text("--");
251 } else {
252 // Si viene como número, mostramos con 1 decimal
253 var tempVal = data["temp"];
254 if (typeof tempVal === "number") {
255 $("#temperature_reading").text(tempVal.toFixed(1) + " °C");
256 } else {
257 $("#temperature_reading").text(tempVal + " °C");
258 }
259 }
260
261 // ----- Estado PIR -----
262 if (data["pir"] === 1) {
263 $("#pir_status").text("Movimiento detectado");
264 } else {
265 $("#pir_status").text("Sin movimiento");
266 }
267 }).fail(function() {
268 $("#temperature_reading").text("--");
269 $("#pir_status").text("Sin datos");
270 });
271}

485/**
486 * @brief Crear o actualizar registro programado
487 * @details Obtiene datos del formulario (número de registro, hora, minuto, días)
488 * y los envía al endpoint REST /api/register mediante POST.
489 *
490 * Los días se envían en formato abreviado: L, M, X, J, V, S, D
491 *
492 * @note El registro se guarda en NVS y persiste tras reinicios
493 */
494function send_register()
495{
496 // Get form values
497 var selectedNumber = parseInt($("#selectNumber").val());
498 var hours = parseInt($("#hours").val());
499 var minutes = parseInt($("#minutes").val());
500
501 // Create array for selected days using new format (L, M, X, J, V, S, D)
502 var selectedDays = [];
503 if ($("#day_mon").prop("checked")) selectedDays.push("L"); // Lunes
504 if ($("#day_tue").prop("checked")) selectedDays.push("M"); // Martes
505 if ($("#day_wed").prop("checked")) selectedDays.push("X"); // Miércoles
506 if ($("#day_thu").prop("checked")) selectedDays.push("J"); // Jueves
507 if ($("#day_fri").prop("checked")) selectedDays.push("V"); // Viernes
508 if ($("#day_sat").prop("checked")) selectedDays.push("S"); // Sábado
509 if ($("#day_sun").prop("checked")) selectedDays.push("D"); // Domingo
510
511 // Create data object for new REST API
512 var requestData = {
513 'register': selectedNumber,
514 'hour': hours,
515 'minute': minutes,
516 'days': selectedDays
517 };
518
519 // Send to new REST API endpoint
520 $.ajax({
521 url: '/api/register',
522 dataType: 'json',
523 method: 'POST',
524 cache: false,
525 data: JSON.stringify(requestData),
526 contentType: 'application/json',
527 success: function(response) {
528 console.log('Register saved successfully:', response);
529 alert('Registro guardado exitosamente');
530 },
531 error: function(xhr, status, error) {
532 console.error('Error saving register:', xhr.responseText);
533 alert('Error al guardar registro');
534 }
535 });
536}

637/**
638 * @brief Aplicar configuración del ventilador
639 * @details Envía modo de operación y velocidad al endpoint /fanControl.json.
640 *
641 * Modos disponibles:
642 * - "manual": Control directo por slider
643 * - "auto": Control automático por temperatura NTC
644 * - "registros": Control por registros programados
645 *
646 * @note En modo auto, la velocidad del slider se usa solo como referencia
647 * @note En modo registros, la velocidad se aplica cuando un registro coincide
648 */
649function apply_fan_control() {
650 const mode = $("#fan_mode").val(); // "manual", "auto", "registros"
651 const speed = parseInt($("#fan_speed").val(), 10);
652
653 const payload = JSON.stringify({
654 mode: mode,
655 speed: speed
656 });
657
658 $.ajax({
659 url: "/fanControl.json",
660 dataType: "json",
661 method: "POST",
662 cache: false,
663 data: payload,
664 contentType: "application/json",
665 success: function(resp) {
666 $("#fan_status").text(resp.status || "Configuración aplicada");
667 },
668 error: function(xhr, status, error) {
669 console.error(xhr.responseText);
