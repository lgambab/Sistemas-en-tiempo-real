233static esp_err_t http_server_fan_control_handler(httpd_req_t *req)
234{
235 ESP_LOGI(TAG, "/fanControl.json requested");
236
237 int len = req->content_len;
238 if (len <= 0) {
239 ESP_LOGI(TAG, "fanControl: content_len <= 0");
240 return ESP_FAIL;
241 }
242
243 char *buf = malloc(len + 1);
244 if (!buf) {
245 ESP_LOGI(TAG, "fanControl: malloc failed");
246 return ESP_FAIL;
247 }
248
249 int ret = httpd_req_recv(req, buf, len);
250 if (ret <= 0) {
251 free(buf);
252 ESP_LOGI(TAG, "fanControl: recv failed");
253 return ESP_FAIL;
254 }
255 buf[len] = '\0';
256
257 cJSON *root = cJSON_Parse(buf);
258 free(buf);
259
260 if (!root) {
261 ESP_LOGI(TAG, "fanControl: invalid JSON");
262 return ESP_FAIL;
263 }
264
265 cJSON *mode_json = cJSON_GetObjectItem(root, "mode");
266 cJSON *speed_json = cJSON_GetObjectItem(root, "speed");
267
268 if (!cJSON_IsString(mode_json) || !cJSON_IsNumber(speed_json)) {
269 cJSON_Delete(root);
270 ESP_LOGI(TAG, "fanControl: missing fields");
271 return ESP_FAIL;
272 }
273
274 const char *mode_str = mode_json->valuestring;
275 int speed = speed_json->valueint;
276
277 fan_mode_t mode = FAN_MODE_MANUAL;
278 if (strcmp(mode_str, "auto") == 0) {
279 mode = FAN_MODE_AUTO;
280 } else if (strcmp(mode_str, "registros") == 0) {
281 mode = FAN_MODE_REGISTERS;
282 }
283
284 fan_control_set_mode(mode);
285 fan_control_set_manual_speed((uint8_t)speed);
286
287 cJSON_Delete(root);
288
289 char resp[128];
290 snprintf(resp, sizeof(resp),
291 "{\"status\":\"ok\",\"mode\":\"%s\",\"speed\":%d}",
292 mode_str, speed);
293
294 httpd_resp_set_type(req, "application/json");
295 httpd_resp_send(req, resp, strlen(resp));
296
297 return ESP_OK;
298}

468static void http_server_monitor(void *parameter)
469{
470 http_server_queue_message_t msg;
471
472 for (;;)
473 {
474 if (xQueueReceive(http_server_monitor_queue_handle, &msg, portMAX_DELAY))
475 {
476 switch (msg.msgID)
477 {
478 case HTTP_MSG_WIFI_CONNECT_INIT:
479 ESP_LOGI(TAG, "HTTP_MSG_WIFI_CONNECT_INIT");
480 g_wifi_connect_status = HTTP_WIFI_STATUS_CONNECTING;
481 break;
482
483 case HTTP_MSG_WIFI_CONNECT_SUCCESS:
484 ESP_LOGI(TAG, "HTTP_MSG_WIFI_CONNECT_SUCCESS");
485 g_wifi_connect_status = HTTP_WIFI_STATUS_CONNECT_SUCCESS;
486 // mqtt_app_start();
487 break;
488
489 case HTTP_MSG_WIFI_CONNECT_FAIL:
490 ESP_LOGI(TAG, "HTTP_MSG_WIFI_CONNECT_FAIL");
491 g_wifi_connect_status = HTTP_WIFI_STATUS_CONNECT_FAILED;
492 break;
493
494 case HTTP_MSG_OTA_UPDATE_SUCCESSFUL:
495 ESP_LOGI(TAG, "HTTP_MSG_OTA_UPDATE_SUCCESSFUL");
496 g_fw_update_status = OTA_UPDATE_SUCCESSFUL;
497 http_server_fw_update_reset_timer();
498 break;
499
500 case HTTP_MSG_OTA_UPDATE_FAILED:
501 ESP_LOGI(TAG, "HTTP_MSG_OTA_UPDATE_FAILED");
502 g_fw_update_status = OTA_UPDATE_FAILED;
503 break;
504
505 default:
506 break;
507 }
508 }
509 }

655esp_err_t http_server_OTA_update_handler(httpd_req_t *req)
656{
657 esp_ota_handle_t ota_handle;
658 char ota_buff[1024];
659 int content_length = req->content_len;
660 int content_received = 0;
661 int recv_len;
662 bool is_req_body_started = false;
663 bool flash_successful = false;
664
665 const esp_partition_t *update_partition = esp_ota_get_next_update_partition(NULL);
666
667 do
668 {
669 if ((recv_len = httpd_req_recv(req, ota_buff, MIN(content_length, sizeof(ota_buff)))) < 0)
670 {
671 if (recv_len == HTTPD_SOCK_ERR_TIMEOUT)
672 {
673 ESP_LOGI(TAG, "http_server_OTA_update_handler: Socket Timeout");
674 continue;
675 }
676 ESP_LOGI(TAG, "http_server_OTA_update_handler: OTA other Error %d", recv_len);
677 return ESP_FAIL;
678 }
679 printf("http_server_OTA_update_handler: OTA RX: %d of %d\r",
680 content_received, content_length);
681
682 if (!is_req_body_started)
683 {
684 is_req_body_started = true;
685
686 char *body_start_p = strstr(ota_buff, "\r\n\r\n") + 4;
687 int body_part_len = recv_len - (body_start_p - ota_buff);
688
689 printf("http_server_OTA_update_handler: OTA file size: %d\r\n", content_length);
690
691 esp_err_t err = esp_ota_begin(update_partition, OTA_SIZE_UNKNOWN, &ota_handle);
692 if (err != ESP_OK)
693 {
694 printf("http_server_OTA_update_handler: Error with OTA begin, cancelling\r\n");
695 return ESP_FAIL;
696 }
697 else
698 {
699 printf("http_server_OTA_update_handler: Writing to partition subtype %d at 0x%lx\r\n",
700 update_partition->subtype, update_partition->address);
701 }
702
703 esp_ota_write(ota_handle, body_start_p, body_part_len);
704 content_received += body_part_len;
705 }
706 else
707 {
708 esp_ota_write(ota_handle, ota_buff, recv_len);
709 content_received += recv_len;
710 }
711
712 } while (recv_len > 0 && content_received < content_length);
713
714 if (esp_ota_end(ota_handle) == ESP_OK)
715 {
716 if (esp_ota_set_boot_partition(update_partition) == ESP_OK)
717 {
718 const esp_partition_t *boot_partition = esp_ota_get_boot_partition();
719 ESP_LOGI(TAG,
720 "http_server_OTA_update_handler: Next boot partition subtype %d at 0x%lx",
721 boot_partition->subtype, boot_partition->address);
722 flash_successful = true;
723 }
724 else
725 {
726 ESP_LOGI(TAG, "http_server_OTA_update_handler: FLASHED ERROR!!!");
727 }
728 }
729 else
730 {
731 ESP_LOGI(TAG, "http_server_OTA_update_handler: esp_ota_end ERROR!!!");
732 }
733
734 if (flash_successful) {
735 http_server_monitor_send_message(HTTP_MSG_OTA_UPDATE_SUCCESSFUL);
736 } else {
737 http_server_monitor_send_message(HTTP_MSG_OTA_UPDATE_FAILED);
738 }
739
740 return ESP_OK;
741}

791static esp_err_t http_server_wifi_connect_json_handler(httpd_req_t *req)
792{
793 size_t header_len;
794 char *header_value;
795 char *ssid_str = NULL;
796 char *pass_str = NULL;
797 int content_length;
798
799 ESP_LOGI(TAG, "/wifiConnect.json requested");
800
801 header_len = httpd_req_get_hdr_value_len(req, "Content-Length");
802 if (header_len <= 0) {
803 ESP_LOGI(TAG, "Content-Length header is missing or invalid");
804 return ESP_FAIL;
805 }
806
807 header_value = (char *)malloc(header_len + 1);
808 if (httpd_req_get_hdr_value_str(req, "Content-Length", header_value, header_len + 1) != ESP_OK) {
809 free(header_value);
810 ESP_LOGI(TAG, "Failed to get Content-Length header value");
811 return ESP_FAIL;
812 }
813
814 content_length = atoi(header_value);
815 free(header_value);
816
817 if (content_length <= 0) {
818 ESP_LOGI(TAG, "Invalid Content-Length value");
819 return ESP_FAIL;
820 }
821
822 char *data_buffer = (char *)malloc(content_length + 1);
823 if (!data_buffer) {
824 ESP_LOGI(TAG, "malloc failed");
825 return ESP_FAIL;
826 }
827
828 if (httpd_req_recv(req, data_buffer, content_length) <= 0) {
829 free(data_buffer);
830 ESP_LOGI(TAG, "Failed to receive request body");
831 return ESP_FAIL;
832 }
833
834 data_buffer[content_length] = '\0';
835
836 cJSON *root = cJSON_Parse(data_buffer);
837 free(data_buffer);
838
839 if (!root) {
840 ESP_LOGI(TAG, "Invalid JSON data");
841 return ESP_FAIL;
842 }
843
844 cJSON *ssid_json = cJSON_GetObjectItem(root, "selectedSSID");
845 cJSON *pwd_json = cJSON_GetObjectItem(root, "pwd");
846
847 if (!ssid_json || !pwd_json || !cJSON_IsString(ssid_json) || !cJSON_IsString(pwd_json)) {
848 cJSON_Delete(root);
849 ESP_LOGI(TAG, "Missing or invalid JSON data fields");
850 return ESP_FAIL;
851 }
852
853 ssid_str = strdup(ssid_json->valuestring);
854 pass_str = strdup(pwd_json->valuestring);
855
856 cJSON_Delete(root);
857
858 ESP_LOGI(TAG, "Received SSID: %s", ssid_str);
859 ESP_LOGI(TAG, "Received Password: %s", pass_str);
860
861 wifi_config_t *wifi_config = wifi_app_get_wifi_config();
862 memset(wifi_config, 0x00, sizeof(wifi_config_t));
863 memcpy(wifi_config->sta.ssid, ssid_str, strlen(ssid_str));
864 memcpy(wifi_config->sta.password, pass_str, strlen(pass_str));
865
866 save_wifi_credentials(ssid_str, pass_str);
867 esp_wifi_disconnect();
868 connect_to_wifi();
869
870 free(ssid_str);
871 free(pass_str);
872
873 return ESP_OK;
874}

994
995 int len = req->content_len;
996 if (len <= 0 || len > 128) {
997 httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, "invalid length");
998 return ESP_FAIL;
999 }
1000
1001 char *buf = malloc(len + 1);
1002 if (!buf) {
1003 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "no mem");
1004 return ESP_FAIL;
1005 }
1006
1007 int ret = httpd_req_recv(req, buf, len);
1008 if (ret <= 0) {
1009 free(buf);
1010 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "recv failed");
1011 return ESP_FAIL;
1012 }
1013 buf[ret] = '\0';
1014
1015 cJSON *root = cJSON_Parse(buf);
1016 free(buf);
1017 if (!root) {
1018 httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, "invalid json");
1019 return ESP_FAIL;
1020 }
1021
1022 cJSON *reg_number_json = cJSON_GetObjectItem(root, "selectedNumber");
1023 if (!cJSON_IsString(reg_number_json)) {
1024 cJSON_Delete(root);
1025 httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, "missing selectedNumber");
1026 return ESP_FAIL;
1027 }
1028
1029 int id = atoi(reg_number_json->valuestring);
1030 cJSON_Delete(root);
1031
1032 if (id <= 0 || id > 10) {
1033 httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, "invalid id");
1034 return ESP_FAIL;
1035 }
1036
1037 delete_register_from_nvs(id);
1038 ESP_LOGI(TAG, "register %d deleted from NVS", id);
1039
1040 httpd_resp_set_type(req, "application/json");
1041 httpd_resp_send(req, "{\"status\":\"ok\"}", -1);
1042
1043 return ESP_OK;
1044}

1252// -----------------------------------------------------
1253// /readreg.json (compatibilidad con front-end viejo)
1254// Devuelve reg1..reg10 como strings "HHMM" + días ("L","M","X"...)
1255// -----------------------------------------------------
1256static esp_err_t http_server_read_register_handler(httpd_req_t *req)
1257{
1258 ESP_LOGI(TAG, "/readreg.json requested -> leyendo desde NVS");
1259
1260 cJSON *root = cJSON_CreateObject();
1261 if (!root) {
1262 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "no mem");
1263 return ESP_FAIL;
1264 }
1265
1266 for (int i = 1; i <= 10; i++) {
1267 char key[16];
1268 snprintf(key, sizeof(key), "reg%d", i);
1269
1270 char buf[256];
1271 esp_err_t err = load_register_from_nvs(i, buf, sizeof(buf));
1272 if (err != ESP_OK) {
1273 // No hay registro: manda null
1274 cJSON_AddNullToObject(root, key);
1275 continue;
1276 }
1277
1278 cJSON *reg_json = cJSON_Parse(buf);
1279 if (!reg_json) {
1280 cJSON_AddNullToObject(root, key);
1281 continue;
1282 }
1283
1284 cJSON *hour_json = cJSON_GetObjectItem(reg_json, "hour");
1285 cJSON *minute_json = cJSON_GetObjectItem(reg_json, "minute");
1286 cJSON *days_json = cJSON_GetObjectItem(reg_json, "days");
1287
1288 if (!cJSON_IsNumber(hour_json) ||
1289 !cJSON_IsNumber(minute_json) ||
1290 !cJSON_IsArray(days_json)) {
1291 cJSON_Delete(reg_json);
1292 cJSON_AddNullToObject(root, key);
1293 continue;
1294 }
1295
1296 char reg_str[32];
1297 // "HHMM"
1298 snprintf(reg_str, sizeof(reg_str), "%02d%02d",
1299 hour_json->valueint, minute_json->valueint);
1300
1301 // Añade letras de días tal como las guardamos en NVS: "L","M","X","J","V","S","D"
1302 int ndays = cJSON_GetArraySize(days_json);
1303 for (int j = 0; j < ndays && strlen(reg_str) < sizeof(reg_str) - 1; j++) {
1304 cJSON *d = cJSON_GetArrayItem(days_json, j);
1305 if (cJSON_IsString(d) && d->valuestring && d->valuestring[0] != '\0') {
1306 strncat(reg_str, d->valuestring, 1); // solo la primera letra
1307 }
1308 }
1309
1310 cJSON_AddStringToObject(root, key, reg_str);
1311 cJSON_Delete(reg_json);
1312 }
1313
1314 char *out = cJSON_PrintUnformatted(root);
1315 httpd_resp_set_type(req, "application/json");
1316 httpd_resp_send(req, out, strlen(out));
1317
1318 free(out);
1319 cJSON_Delete(root);
1320
1321 return ESP_OK;
1322}
