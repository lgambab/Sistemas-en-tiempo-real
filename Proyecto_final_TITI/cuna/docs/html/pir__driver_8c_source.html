102void pir_init(gpio_num_t pir_gpio, QueueHandle_t pir_queue)
103{
104 s_pir_gpio = pir_gpio;
105 s_pir_evt_queue = pir_queue;
106
107 // Configuración del GPIO como entrada con detección de flancos
108 gpio_config_t io_conf = {
109 .pin_bit_mask = 1ULL << s_pir_gpio, // Máscara de bits para el pin
110 .mode = GPIO_MODE_INPUT, // Modo entrada
111 .pull_up_en = GPIO_PULLUP_DISABLE, // Sin pull-up (sensor lo incluye)
112 .pull_down_en = GPIO_PULLDOWN_DISABLE, // Sin pull-down
113 .intr_type = GPIO_INTR_ANYEDGE // Interrupción en ambos flancos
114 // Alternativa: GPIO_INTR_POSEDGE para solo flanco de subida
115 };
116
117 gpio_config(&io_conf);
118
119 // Instalar servicio global de ISR (solo primera vez)
120 // Flags=0 usa valores por defecto
121 gpio_install_isr_service(0);
122
123 // Registrar ISR específica para este pin
124 gpio_isr_handler_add(s_pir_gpio, pir_isr_handler, NULL);
125}
