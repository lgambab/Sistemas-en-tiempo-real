149static void obtain_time(void)
150{
151 setenv("TZ", "EST5EDT,M3.2.0/2,M11.1.0", 1);
152 tzset();
153 ESP_LOGI(TAG, "Initializing SNTP");
154
155 // Stop SNTP if already running to avoid assert
156 if (esp_sntp_enabled()) {
157 ESP_LOGI(TAG, "SNTP already running, stopping first");
158 esp_sntp_stop();
159 }
160
161 // Configurar el servidor SNTP. Aquí se utiliza "pool.ntp.org" como ejemplo. Puedes cambiarlo según tus necesidades.
162 esp_sntp_setoperatingmode(SNTP_OPMODE_POLL);
163 esp_sntp_setservername(0, "0.co.pool.ntp.org");
164 esp_sntp_init();
165
166 // Esperar a que se sincronice el tiempo con el servidor SNTP
167 time_t now = 0;
168 struct tm timeinfo = {0};
169 int retry = 0;
170 const int retry_count = 10;
171
172 while (timeinfo.tm_year < (2016 - 1900) && ++retry < retry_count)
173 {
174 ESP_LOGI(TAG, "Waiting for system time to be set... (%d/%d)", retry, retry_count);
175 vTaskDelay(2000 / portTICK_PERIOD_MS);
176 time(&now);
177 localtime_r(&now, &timeinfo);
178 }
179
180 if (retry < retry_count)
181 {
182 ESP_LOGI(TAG, "System time is set!");
183 time_was_synchronized = true;
184 }
185 else
186 {
187 ESP_LOGE(TAG, "Unable to set system time. Check your SNTP configuration.");
188 }
189}

508void load_wifi_credentials(char *ssid, char *password) {
509 nvs_handle_t nvs_handle;
510 ESP_ERROR_CHECK(nvs_open("storage", NVS_READONLY, &nvs_handle));
511
512 size_t required_size;
513
514 // Get the size of wifi_ssid
515 ESP_ERROR_CHECK(nvs_get_str(nvs_handle, "wifi_ssid", NULL, &required_size));
516 // Allocate memory for wifi_ssid
517 char *ssid_buffer = malloc(required_size);
518 if (ssid_buffer == NULL) {
519 // Handle memory allocation error
520 ESP_LOGE(TAG, "Failed to allocate memory for wifi_ssid");
521 nvs_close(nvs_handle);
522 return;
523 }
524 // Get wifi_ssid
525 ESP_ERROR_CHECK(nvs_get_str(nvs_handle, "wifi_ssid", ssid_buffer, &required_size));
526 // Copy wifi_ssid to the output parameter
527 strncpy(ssid, ssid_buffer, required_size);
528
529 // Repeat the process for wifi_password
530 ESP_ERROR_CHECK(nvs_get_str(nvs_handle, "wifi_password", NULL, &required_size));
531 char *password_buffer = malloc(required_size);
532 if (password_buffer == NULL) {
533 // Handle memory allocation error
534 ESP_LOGE(TAG, "Failed to allocate memory for wifi_password");
535 free(ssid_buffer);
536 nvs_close(nvs_handle);
537 return;
538 }
539 ESP_ERROR_CHECK(nvs_get_str(nvs_handle, "wifi_password", password_buffer, &required_size));
540 strncpy(password, password_buffer, required_size);
541
542 // Free the allocated memory
543 free(ssid_buffer);
544 free(password_buffer);
545
546 nvs_close(nvs_handle);
547}

561
562 err = nvs_get_str(nvs_handle, "wifi_password", NULL, &password_size);
563 nvs_close(nvs_handle);

565 return err == ESP_OK;

792static void wifi_app_soft_ap_config(void)
793{
794 // SoftAP - WiFi access point configuration
795 wifi_config_t ap_config =
796 {
797 .ap = {
798 .ssid = WIFI_AP_SSID,
799 .ssid_len = strlen(WIFI_AP_SSID),
800 .password = WIFI_AP_PASSWORD,
801 .channel = WIFI_AP_CHANNEL,
802 .ssid_hidden = WIFI_AP_SSID_HIDDEN,
803 .authmode = WIFI_AUTH_WPA2_PSK,
804 .max_connection = WIFI_AP_MAX_CONNECTIONS,
805 .beacon_interval = WIFI_AP_BEACON_INTERVAL,
806 },
807 };
808
809 // Configure DHCP for the AP
810 esp_netif_ip_info_t ap_ip_info;
811 memset(&ap_ip_info, 0x00, sizeof(ap_ip_info));
812
813 esp_netif_dhcps_stop(esp_netif_ap);
814 inet_pton(AF_INET, WIFI_AP_IP, &ap_ip_info.ip);
815 inet_pton(AF_INET, WIFI_AP_GATEWAY, &ap_ip_info.gw);
816 inet_pton(AF_INET, WIFI_AP_NETMASK, &ap_ip_info.netmask);
817 ESP_ERROR_CHECK(esp_netif_set_ip_info(esp_netif_ap, &ap_ip_info));
818 ESP_ERROR_CHECK(esp_netif_dhcps_start(esp_netif_ap));
819
820 ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_APSTA));
