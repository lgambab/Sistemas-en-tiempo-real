137esp_err_t save_register_to_nvs(int id, reg_t *reg)
138{
139 nvs_handle_t nvs;
140 char key[16];
141 sprintf(key, "reg_%d", id);
142
143 cJSON *root = cJSON_CreateObject();
144 cJSON_AddNumberToObject(root, "hour", reg->hour);
145 cJSON_AddNumberToObject(root, "minute", reg->minute);
146
147 cJSON *days = cJSON_CreateArray();
148 for (int i = 0; i < reg->day_count; i++) {
149 cJSON_AddItemToArray(days, cJSON_CreateString(reg->days[i]));
150 }
151 cJSON_AddItemToObject(root, "days", days);
152
153 char *json_str = cJSON_Print(root);
154
155 ESP_LOGI(TAG, "save_register_to_nvs: id=%d json=%s", id, json_str);
156
157 ESP_ERROR_CHECK(nvs_open("storage", NVS_READWRITE, &nvs));
158 esp_err_t err = nvs_set_str(nvs, key, json_str);
159 nvs_commit(nvs);
160 nvs_close(nvs);
161
162 free(json_str);
163 cJSON_Delete(root);
164
165 return err;
166}

192esp_err_t load_register_from_nvs(int id, char *buffer, size_t buffer_len)

194 nvs_handle_t nvs;

197
198 ESP_ERROR_CHECK(nvs_open("storage", NVS_READWRITE, &nvs));

211 }
212
213 err = nvs_get_str(nvs, key, buffer, &required_size);
214 nvs_close(nvs);

265static esp_err_t load_register_struct_from_nvs(int id, reg_t *out)
266{
267 char buf[512];
268 esp_err_t err = load_register_from_nvs(id, buf, sizeof(buf));
269 if (err != ESP_OK) {
270 return err;
271 }
272
273 cJSON *root = cJSON_Parse(buf);
274 if (!root) {
275 return ESP_FAIL;
276 }
277
278 cJSON *hour_json = cJSON_GetObjectItem(root, "hour");
279 cJSON *min_json = cJSON_GetObjectItem(root, "minute");
280 cJSON *days_json = cJSON_GetObjectItem(root, "days");
281
282 if (!cJSON_IsNumber(hour_json) || !cJSON_IsNumber(min_json) || !cJSON_IsArray(days_json)) {
283 cJSON_Delete(root);
284 return ESP_FAIL;
285 }
286
287 memset(out, 0, sizeof(*out));
288 out->hour = hour_json->valueint;
289 out->minute = min_json->valueint;
290 out->day_count = cJSON_GetArraySize(days_json);
291 if (out->day_count > 7) {
292 out->day_count = 7;
293 }
294
295 for (int i = 0; i < out->day_count; i++) {
296 cJSON *item = cJSON_GetArrayItem(days_json, i);
297 if (cJSON_IsString(item) && item->valuestring) {
298 strncpy(out->days[i], item->valuestring, sizeof(out->days[i]) - 1);
299 out->days[i][sizeof(out->days[i]) - 1] = '\0';
300 } else {
301 out->days[i][0] = '\0';
302 }
303 }
304
305 cJSON_Delete(root);
306 return ESP_OK;
307}

338static bool register_is_active_now(const reg_t *reg)
339{
340 time_t now;
341 struct tm timeinfo;
342 time(&now);
343 localtime_r(&now, &timeinfo);
344
345 int cur_hour = timeinfo.tm_hour;
346 int cur_min = timeinfo.tm_min;
347 int cur_wday = timeinfo.tm_wday; // 0=domingo, 1=lunes, ... 6=sábado
348
349 if (reg->hour != cur_hour || reg->minute != cur_min) {
350 return false;
351 }
352
353 for (int i = 0; i < reg->day_count; i++) {
354 const char *d = reg->days[i];
355
356 if ((cur_wday == 1 && strcmp(d, "L") == 0) || // lunes
357 (cur_wday == 2 && strcmp(d, "M") == 0) || // martes
358 (cur_wday == 3 && strcmp(d, "X") == 0) || // miércoles
359 (cur_wday == 4 && strcmp(d, "J") == 0) || // jueves
360 (cur_wday == 5 && strcmp(d, "V") == 0) || // viernes
361 (cur_wday == 6 && strcmp(d, "S") == 0) || // sábado
362 (cur_wday == 0 && strcmp(d, "D") == 0)) { // domingo
363 return true;
364 }
365 }
366
367 return false;
368}

371esp_err_t api_post_register(httpd_req_t *req)
372{
373 int len = req->content_len;
374 if (len <= 0 || len > 4096) {
375 httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, "invalid length");
376 return ESP_FAIL;
377 }
378
379 char *buf = malloc(len + 1);
380 if (!buf) {
381 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "no mem");
382 return ESP_FAIL;
383 }
384
385 int ret = httpd_req_recv(req, buf, len);
386 if (ret <= 0) {
387 free(buf);
388 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "recv failed");
389 return ESP_FAIL;
390 }
391 buf[ret] = '\0';
392
393 cJSON *root = cJSON_Parse(buf);
394 free(buf);
395
396 if (!root) {
397 httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, "invalid json");
398 return ESP_FAIL;
399 }
400
401 cJSON *id_json = cJSON_GetObjectItem(root, "register");
402 cJSON *hour_json = cJSON_GetObjectItem(root, "hour");
403 cJSON *min_json = cJSON_GetObjectItem(root, "minute");
404 cJSON *days = cJSON_GetObjectItem(root, "days");
405
406 if (!cJSON_IsNumber(id_json) || !cJSON_IsNumber(hour_json) ||
407 !cJSON_IsNumber(min_json) || !cJSON_IsArray(days)) {
408 cJSON_Delete(root);
409 httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, "missing fields");
410 return ESP_FAIL;
411 }
412
413 int id = id_json->valueint;
414 int hour = hour_json->valueint;
415 int minute = min_json->valueint;
416
417 int day_count = cJSON_GetArraySize(days);
418 if (day_count > 7) day_count = 7;
419
420 reg_t reg = {0};
421 reg.hour = hour;
422 reg.minute = minute;
423 reg.day_count = day_count;
424
425 for (int i = 0; i < day_count; i++) {
426 cJSON *item = cJSON_GetArrayItem(days, i);
427 if (cJSON_IsString(item) && item->valuestring) {
428 strncpy(reg.days[i], item->valuestring, sizeof(reg.days[i]) - 1);
429 reg.days[i][sizeof(reg.days[i]) - 1] = '\0';
430 } else {
431 reg.days[i][0] = '\0';
432 }
433 }
434
435 esp_err_t err = save_register_to_nvs(id, &reg);
436 cJSON_Delete(root);
437
438 if (err != ESP_OK) {
439 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "nvs error");
440 return ESP_FAIL;
441 }
442
443 ESP_LOGI(TAG, "api_post_register: saved id=%d hour=%d minute=%d days=%d",
444 id, hour, minute, day_count);
445
446 httpd_resp_sendstr(req, "OK");
447 return ESP_OK;
448}

450// -------------------- HTTP: GET /api/registers --------------------
451esp_err_t api_get_registers(httpd_req_t *req)
452{
453 cJSON *root = cJSON_CreateObject();
454 char smallbuf[256];
455
456 for (int i = 1; i <= 10; i++) {
457 char key[8];
458 snprintf(key, sizeof(key), "%d", i);
459
460 esp_err_t r = load_register_from_nvs(i, smallbuf, sizeof(smallbuf));
461 if (r == ESP_OK) {
462 cJSON *reg_json = cJSON_Parse(smallbuf);
463 if (reg_json) {
464 cJSON_AddItemToObject(root, key, reg_json);
465 continue;
466 }
467 } else if (r == ESP_ERR_NVS_INVALID_LENGTH) {
468 nvs_handle_t nvs;
469 char nkey[16];
470 snprintf(nkey, sizeof(nkey), "reg_%d", i);
471 if (nvs_open("storage", NVS_READWRITE, &nvs) == ESP_OK) {
472 size_t required = 0;
473 if (nvs_get_str(nvs, nkey, NULL, &required) == ESP_OK && required > 0) {
474 char *dyn = malloc(required);
475 if (dyn) {
476 if (nvs_get_str(nvs, nkey, dyn, &required) == ESP_OK) {
477 cJSON *reg_json = cJSON_Parse(dyn);
478 if (reg_json) {
479 cJSON_AddItemToObject(root, key, reg_json);
480 }
481 }
482 free(dyn);
483 }
484 }
485 nvs_close(nvs);
486 continue;
487 }
488 }
489
490 cJSON_AddNullToObject(root, key);
491 }
492
493 char *out = cJSON_PrintUnformatted(root);
494
495 httpd_resp_set_type(req, "application/json");
496 httpd_resp_send(req, out, strlen(out));
497
498 free(out);
499 cJSON_Delete(root);
500 return ESP_OK;

526
